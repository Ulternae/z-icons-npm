# @zcorvus/core-react

Standalone React package for one ZCorvus Core icon variant.

```tsx
import { CoreAirplay, ZCore, ZCoreSprite } from '@zcorvus/core-react';

export function Example() {
  return (
    <>
      <CoreAirplay />
      <ZCoreSprite />
      <ZCore name="crown" />
    </>
  );
}
```
