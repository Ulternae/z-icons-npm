# @zcorvus/core-solid-react

Standalone React package for one ZCorvus Core icon variant.

```tsx
import { CoreAirplaySolid, ZCoreSolid, ZCoreSolidSprite } from '@zcorvus/core-solid-react';

export function Example() {
  return (
    <>
      <CoreAirplaySolid />
      <ZCoreSolidSprite />
      <ZCoreSolid name="airplay" />
    </>
  );
}
```
